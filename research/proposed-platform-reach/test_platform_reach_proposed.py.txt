"""Platform reach (research/limits-platform-reach-2026-10-05.md): the device-free fixes. No simulator, emulator or SDK is touched.

Run: python3 -m unittest tests.test_platform_reach
"""
import json, os, subprocess, sys, tempfile, unittest
from pathlib import Path
from unittest import mock

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))
import shotfleet as sf


def types(*names):
    return json.dumps({"devicetypes": [{"name": n, "identifier": "com.apple.CoreSimulator.SimDeviceType." + n.replace(" ", "-").replace("(", "").replace(")", "")} for n in names]})


class DefaultDeviceType(unittest.TestCase):
    def test_xcode_with_the_iphone_17_pro_max_keeps_the_old_default(self):
        self.assertEqual(sf.default_device_type(types("iPhone 16 Pro Max", "iPhone 17 Pro Max", "iPad Pro 13-inch (M5)")), sf.DEFAULT_IPHONE)

    def test_an_older_xcode_gets_its_newest_pro_max_not_a_missing_type(self):
        listed = types("iPhone 15 Pro Max", "iPhone 16 Pro Max", "iPhone 16 Pro", "iPhone 9 Pro Max", "iPad Pro 13-inch (M4)")
        self.assertEqual(sf.default_device_type(listed), "com.apple.CoreSimulator.SimDeviceType.iPhone-16-Pro-Max")  # 16, not the string-sorted 9

    def test_nothing_usable_or_unreadable_falls_back_to_the_old_default_and_simctl_names_the_problem(self):
        for listed in (types("iPad Pro 13-inch (M4)"), "", "not json", "[]"):
            self.assertEqual(sf.default_device_type(listed), sf.DEFAULT_IPHONE)

    def test_it_asks_simctl_once_with_a_timeout_and_survives_a_dead_simctl(self):
        seen = []
        def ok(cmd, **kw): seen.append((cmd, kw.get("timeout"))); return subprocess.CompletedProcess(cmd, 0, types("iPhone 16 Pro Max"), "")
        with mock.patch.object(sf, "sh", ok):
            self.assertTrue(sf.default_device_type().endswith("iPhone-16-Pro-Max"))
        self.assertEqual(seen, [(["xcrun", "simctl", "list", "devicetypes", "-j"], 60)])
        with mock.patch.object(sf, "sh", side_effect=subprocess.TimeoutExpired("x", 60)):
            self.assertEqual(sf.default_device_type(), sf.DEFAULT_IPHONE)
        with mock.patch.object(sf, "sh", side_effect=FileNotFoundError("xcrun")):
            self.assertEqual(sf.default_device_type(), sf.DEFAULT_IPHONE)


class OlderAndroidThan33(unittest.TestCase):
    """An API 31 emulator would boot (minutes), then fail at `cmd locale set-app-locales`. Refuse before booting."""
    def setUp(self):
        tmp = tempfile.TemporaryDirectory(); self.addCleanup(tmp.cleanup)
        self.avds = Path(tmp.name)
        for p in (mock.patch.object(sf, "avd_home", return_value=self.avds), mock.patch.object(sf, "java_env", return_value={}),
                  mock.patch.object(sf, "start_emu", side_effect=self.Booted)):
            p.start(); self.addCleanup(p.stop)

    class Booted(Exception): pass

    def avd(self, name, sysdir):
        d = self.avds / f"{name}.avd"; d.mkdir(); (d / "config.ini").write_text(f"hw.lcd.width=1080\nimage.sysdir.1={sysdir}\n")

    def test_a_config_image_older_than_33_is_refused_before_avdmanager_runs(self):
        with mock.patch.object(sf, "sh") as shell, self.assertRaises(sf.Fail) as e:
            sf.android_pool(1, {"system_image": "system-images;android-31;google_apis;x86_64"}, [])
        shell.assert_not_called()
        msg = str(e.exception)
        for part in ("API 31", "set-app-locales", "API 33", "minSdk", "system_image"): self.assertIn(part, msg)

    def test_an_existing_emulator_on_an_old_image_is_refused_by_name(self):
        self.avd("shotfleet_android", "system-images/android-30/google_apis/arm64-v8a/")
        with self.assertRaises(sf.Fail) as e: sf.android_pool(1, {}, [])
        self.assertIn("the emulator shotfleet_android is Android API 30", str(e.exception))

    def test_33_and_newer_images_and_unreadable_configs_go_on_to_boot(self):
        for sysdir in ("system-images/android-33/google_apis/x86_64/", "system-images/android-36.1/google_apis/arm64-v8a/"):
            d = self.avds / "shotfleet_android.avd"
            if d.exists(): (d / "config.ini").unlink(); d.rmdir()
            self.avd("shotfleet_android", sysdir)
            with self.assertRaises(self.Booted): sf.android_pool(1, {}, [])  # got past the check; the fake emulator start stops it
        (self.avds / "shotfleet_android.avd/config.ini").unlink()           # a user's AVD without a readable config: not our call to refuse
        with self.assertRaises(self.Booted): sf.android_pool(1, {}, [])


def load_minos_tool():
    import importlib.util
    spec = importlib.util.spec_from_file_location("macho_minos", ROOT / "tools/macho-minos.py")
    mod = importlib.util.module_from_spec(spec); spec.loader.exec_module(mod)
    return mod


BUILD = "Load command 11\n      cmd LC_BUILD_VERSION\n  cmdsize 32\n platform {p}\n    minos {v}\n      sdk 27.0\n   ntools 1\n"


class MachoMinos(unittest.TestCase):
    """tools/macho-minos.py: which macOS a PyInstaller folder needs (Homebrew's Python made the 0.1.0 folder need macOS 27.0)."""
    def setUp(self): self.tool = load_minos_tool()

    def test_reads_the_minimum_macos_of_a_build_version_command(self):
        self.assertEqual(self.tool.minos(BUILD.format(p=1, v="13.0")), (13, 0))
        self.assertEqual(self.tool.minos(BUILD.format(p="MACOS", v="26.4")), (26, 4))

    def test_an_ios_simulator_file_has_no_macos_minimum_and_old_files_use_the_old_command(self):
        self.assertIsNone(self.tool.minos(BUILD.format(p=7, v="14.0")))  # platform 7: iOS simulator; its minos is an iOS version
        self.assertEqual(self.tool.minos("      cmd LC_VERSION_MIN_MACOSX\n  cmdsize 16\n  version 10.9\n      sdk 12.0\n"), (10, 9))
        self.assertIsNone(self.tool.minos(""))

    def test_max_turns_a_too_new_file_into_exit_code_1_and_the_report_names_the_macos(self):
        import contextlib, io
        found = {("arm64", (11, 0)): [Path("a")], ("arm64", (27, 0)): [Path("b"), Path("c")]}
        with mock.patch.object(self.tool, "scan", return_value=found):
            for argv, code in (([], 0), (["--max", "14"], 1), (["--max", "27"], 0)):
                out = io.StringIO()
                with contextlib.redirect_stdout(out): self.assertEqual(self.tool.main(["dist/x", *argv]), code, argv)
            self.assertIn("this build needs macOS 27.0 or newer", out.getvalue()); self.assertIn("2 file(s)", out.getvalue())

    def test_the_build_script_aims_the_ocr_helper_at_macos_13_and_can_use_another_python(self):
        s = (ROOT / "packaging/build.sh").read_text()
        self.assertIn('-target "$ARCH-apple-macos${MACOS_MIN:-13.0}" ocr.swift', s)
        self.assertIn('"${PYTHON:-python3}" -m venv build/venv', s)
        self.assertIn("tools/macho-minos.py dist/shotfleet", s)


if __name__ == "__main__":
    unittest.main()
